ALTER PROCEDURE [dbo].[SP_SaveServerSettings]
(
    @Task NVARCHAR(10), -- INSERT / UPDATE

    @ID INT = NULL OUTPUT,
    @InputFilePath NVARCHAR(MAX) = NULL,
    @OutputFilePath NVARCHAR(MAX) = NULL,
    @BackupFilePath NVARCHAR(MAX) = NULL,
    @NonConvertedFile NVARCHAR(MAX) = NULL,
    @ErrorLog NVARCHAR(MAX) = NULL,
    @AuditLog NVARCHAR(MAX) = NULL,
    @BlankExcel NVARCHAR(MAX) = NULL,
    @Document_Release_Template NVARCHAR(MAX) = NULL,
    @Document_Release NVARCHAR(MAX) = NULL,
    @Payment_Rejection NVARCHAR(MAX) = NULL,
    @Payment_Rejection_Date NVARCHAR(MAX) = NULL,
    @TRADE_EMAIL NVARCHAR(2000) = NULL,
    @Frequency NVARCHAR(50) = NULL,
    @INV_CONF_EMAIL NVARCHAR(2000) = NULL,
    @INV_CONF_FNAME NVARCHAR(200) = NULL,
    @ORD_MIS_EMAIL NVARCHAR(2000) = NULL,
    @PAYREC_TRADE_EMAIL NVARCHAR(2000) = NULL,
    @PHY_INV NVARCHAR(2000) = NULL,
    @NO_INV NVARCHAR(2000) = NULL,
    @EOD_MIS NVARCHAR(2000) = NULL,
    @ORD_DEL NVARCHAR(2000) = NULL,
    @INV_PHY_NTRC NVARCHAR(2000) = NULL,
    @IntraDayPath NVARCHAR(MAX) = NULL,
    @EOD_File_EMail NVARCHAR(2000) = NULL,
    @MISOutputFilePath NVARCHAR(MAX) = NULL,
    @FCC_EMail NVARCHAR(2000) = NULL,
    @Payment_Rejection_EMail NVARCHAR(2000) = NULL,
    @DO_Cancel_Email NVARCHAR(2000) = NULL,
    @DO_Invoice_Cancel_Email NVARCHAR(2000) = NULL,
    @Invoice_Cancel_Email NVARCHAR(2000) = NULL,
    @Fail_MSIL NVARCHAR(2000) = NULL,
    @Sleep_Time_in_Mint INT = NULL,
    @Confirmation_Mail NVARCHAR(2000) = NULL,
    @SMTP_HOST NVARCHAR(200) = NULL,
    @Port INT = NULL,
    @Email_FromID NVARCHAR(200) = NULL,
    @UserID NVARCHAR(100) = NULL,
    @Password NVARCHAR(200) = NULL,
    @Sys_Email_FromID NVARCHAR(200) = NULL,
    @Sys_PWD NVARCHAR(200) = NULL,
	@IR_Mail_Time NVARCHAR(10) = NULL,
	@Auto_IR_prcoess NVARCHAR(3) = NULL,
	@PMT_Host NVARCHAR(50) = NULL,
	@PMT_Port NVARCHAR(50) = NULL,
	@PMT_UserName NVARCHAR(200) = NULL,
	@PMT_Password NVARCHAR(200) = NULL,
	@PMT_RemoteInFolder NVARCHAR(200) = NULL,
	@PMT_RemoteArchieveFolder NVARCHAR(200) = NULL,
	@PMT_LocalFolder NVARCHAR(200) = NULL
)
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY
        BEGIN TRANSACTION;

        IF (@Task = 'Save')
        BEGIN
            INSERT INTO Settings_Server
            (
                InputFilePath, OutputFilePath, BackupFilePath, NonConvertedFile,   ErrorLog, AuditLog, BlankExcel, Document_Release_Template,
                Document_Release, Payment_Rejection, Payment_Rejection_Date,  TRADE_EMAIL, Frequency, INV_CONF_EMAIL, INV_CONF_FNAME,
                ORD_MIS_EMAIL, PAYREC_TRADE_EMAIL, PHY_INV, NO_INV,   EOD_MIS, ORD_DEL, INV_PHY_NTRC, IntraDayPath,  EOD_File_EMail, 
				MISOutputFilePath, FCC_EMail,  Payment_Rejection_EMail, DO_Cancel_Email,   DO_Invoice_Cancel_Email, Invoice_Cancel_Email,
                Fail_MSIL, Sleep_Time_in_Mint, Confirmation_Mail,   SMTP_HOST, Port, Email_FromID, UserID, Password,  Sys_Email_FromID, Sys_PWD,IR_Mail_Time, PMT_Host, PMT_Port, PMT_UserName, PMT_Password, PMT_RemoteInFolder, PMT_RemoteArchieveFolder, PMT_LocalFolder  )
            VALUES
            (
                @InputFilePath, @OutputFilePath, @BackupFilePath, @NonConvertedFile,    @ErrorLog, @AuditLog, @BlankExcel, @Document_Release_Template,
                @Document_Release, @Payment_Rejection, @Payment_Rejection_Date, @TRADE_EMAIL, @Frequency, @INV_CONF_EMAIL, @INV_CONF_FNAME,
                @ORD_MIS_EMAIL, @PAYREC_TRADE_EMAIL, @PHY_INV, @NO_INV,    @EOD_MIS, @ORD_DEL, @INV_PHY_NTRC, @IntraDayPath,    @EOD_File_EMail, 
				@MISOutputFilePath, @FCC_EMail,  @Payment_Rejection_EMail, @DO_Cancel_Email,  @DO_Invoice_Cancel_Email,
				@Invoice_Cancel_Email,    @Fail_MSIL, @Sleep_Time_in_Mint, @Confirmation_Mail,   @SMTP_HOST, @Port, @Email_FromID, @UserID, 
				@Password,  @Sys_Email_FromID, @Sys_PWD  ,@IR_Mail_Time
				, @PMT_Host, @PMT_Port, @PMT_UserName, @PMT_Password, @PMT_RemoteInFolder, @PMT_RemoteArchieveFolder, @PMT_LocalFolder);

            SET @ID = SCOPE_IDENTITY();
            
            select 'Record inserted successfully';
        END

        ELSE IF (@Task = 'Edit')
        BEGIN
            UPDATE Settings_Server
            SET
                InputFilePath = @InputFilePath,
                OutputFilePath = @OutputFilePath,
                BackupFilePath = @BackupFilePath,
                NonConvertedFile = @NonConvertedFile,
                ErrorLog = @ErrorLog,
                AuditLog = @AuditLog,
                BlankExcel = @BlankExcel,
                Document_Release_Template = @Document_Release_Template,
                Document_Release = @Document_Release,
                Payment_Rejection = @Payment_Rejection,
                Payment_Rejection_Date = @Payment_Rejection_Date,
                TRADE_EMAIL = @TRADE_EMAIL,
                Frequency = @Frequency,
                INV_CONF_EMAIL = @INV_CONF_EMAIL,
                INV_CONF_FNAME = @INV_CONF_FNAME,
                ORD_MIS_EMAIL = @ORD_MIS_EMAIL,
                PAYREC_TRADE_EMAIL = @PAYREC_TRADE_EMAIL,
                PHY_INV = @PHY_INV,
                NO_INV = @NO_INV,
                EOD_MIS = @EOD_MIS,
                ORD_DEL = @ORD_DEL,
                INV_PHY_NTRC = @INV_PHY_NTRC,
                IntraDayPath = @IntraDayPath,
                EOD_File_EMail = @EOD_File_EMail,
                MISOutputFilePath = @MISOutputFilePath,
                FCC_EMail = @FCC_EMail,
                
                Payment_Rejection_EMail = @Payment_Rejection_EMail,
                DO_Cancel_Email = @DO_Cancel_Email,
                DO_Invoice_Cancel_Email = @DO_Invoice_Cancel_Email,
                Invoice_Cancel_Email = @Invoice_Cancel_Email,
                Fail_MSIL = @Fail_MSIL,
                Sleep_Time_in_Mint = @Sleep_Time_in_Mint,
                Confirmation_Mail = @Confirmation_Mail,
                SMTP_HOST = @SMTP_HOST,
                Port = @Port,
                Email_FromID = @Email_FromID,
                UserID = @UserID,
                Password = @Password,
                Sys_Email_FromID = @Sys_Email_FromID,
                Sys_PWD = @Sys_PWD,
				IR_Mail_Time=@IR_Mail_Time,
				PMT_Host=@PMT_Host, PMT_Port=@PMT_Port, PMT_UserName=@PMT_UserName, PMT_Password=@PMT_Password, 
				PMT_RemoteInFolder=@PMT_RemoteInFolder, PMT_RemoteArchieveFolder=@PMT_RemoteArchieveFolder, PMT_LocalFolder=@PMT_LocalFolder
            WHERE ID = @ID;

            IF @@ROWCOUNT = 0
            BEGIN
                
               select 'No record found to update';
            END
            ELSE
            BEGIN

                select 'Record updated successfully';
            END
        END

        COMMIT TRANSACTION;
    END TRY

    BEGIN CATCH
        ROLLBACK TRANSACTION;

        
        select ERROR_MESSAGE();
    END CATCH
END
